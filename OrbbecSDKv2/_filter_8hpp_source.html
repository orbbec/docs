61template <typename T> T getPropertyRange(const OBFilterConfigSchemaItem &item, const double cur) {
62 // If T type is illegal, T will be void
63 using valueType = typename RangeTraits<T>::valueType;
64 T range{};
65 // Compilate error will be reported here if T is void
66 range.cur = static_cast<valueType>(cur);
67 range.def = static_cast<valueType>(item.def);
68 range.max = static_cast<valueType>(item.max);
69 range.min = static_cast<valueType>(item.min);
70 range.step = static_cast<valueType>(item.step);
71 return range;
72}

90 virtual void init(ob_filter *impl) {
91 impl_ = impl;
92 ob_error *error = nullptr;
93 name_ = ob_filter_get_name(impl_, &error);
94 Error::handle(&error);
95
96 auto configSchemaList = ob_filter_get_config_schema_list(impl_, &error);
97 Error::handle(&error);
98
99 auto count = ob_filter_config_schema_list_get_count(configSchemaList, &error);
100 Error::handle(&error);
101
102 for(uint32_t i = 0; i < count; i++) {
103 auto item = ob_filter_config_schema_list_get_item(configSchemaList, i, &error);
104 Error::handle(&error);
105 configSchemaVec_.emplace_back(item);
106 }
107
108 ob_delete_filter_config_schema_list(configSchemaList, &error);
109 Error::handle(&error);
110 }
