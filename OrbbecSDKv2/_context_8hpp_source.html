232 void setDeviceChangedCallback(DeviceChangedCallback callback) {
233 std::lock_guard<std::mutex> lock(legacyCallbackMtx_);
234 // remove the last callback first
235 if(legacyCallbackId_ != INVALID_CALLBACK_ID) {
236 unregisterDeviceChangedCallback(legacyCallbackId_);
237 }
238 // register the new callback
239 legacyCallbackId_ = registerDeviceChangedCallback(callback);
240 }
241
242 OBCallbackId registerDeviceChangedCallback(DeviceChangedCallback callback) {
243 ob_error *error = nullptr;
244 auto cbCtxHolder = std::unique_ptr<DeviceChangedCallbackContext>(new DeviceChangedCallbackContext({ this, 0, nullptr }));
245 auto id = ob_register_device_changed_callback(impl_, &Context::deviceChangedCallback, (void *)cbCtxHolder.get(), &error);
246 Error::handle(&error);
247
248 cbCtxHolder->callbackId = id;
249 cbCtxHolder->callback = callback;
250 {
251 std::unique_lock<std::mutex> lock(callbackMtx_);
252 devChangedCallbacks_[id] = std::move(cbCtxHolder);
253 }
254 return id;
255 }
256
257 void unregisterDeviceChangedCallback(OBCallbackId id) {
258 ob_error *error = nullptr;
259 ob_unregister_device_changed_callback(impl_, id, &error);
260 {
261 std::unique_lock<std::mutex> lock(callbackMtx_);
262 if(devChangedCallbacks_.count(id)) {
263 devChangedCallbacks_.erase(id);
264 }
265 }
266 Error::handle(&error);
267 }

409 static void deviceChangedCallback(ob_device_list *removedList, ob_device_list *addedList, void *userData) {
410 auto cbCtx = static_cast<DeviceChangedCallbackContext *>(userData);
411 if(cbCtx && cbCtx->ctx && cbCtx->callbackId != INVALID_CALLBACK_ID) {
412 DeviceChangedCallback callbackCopy;
413 {
414 // preventing the captured variables from being released.
415 std::unique_lock<std::mutex> lock(cbCtx->ctx->callbackMtx_);
416 if(cbCtx->ctx->devChangedCallbacks_.count(cbCtx->callbackId)) {
417 callbackCopy = cbCtx->callback;
418 }
419 }
420 auto removed = std::make_shared<DeviceList>(removedList);
421 auto added = std::make_shared<DeviceList>(addedList);
422 callbackCopy(removed, added);
423 }
424 }

426 static void logCallback(OBLogSeverity severity, const char *logMsg, void *userData) {
427 auto cb = static_cast<LogCallback *>(userData);
428 if(cb) {
429 (*cb)(severity, logMsg);
430 }
431 }
