221 std::vector<std::shared_ptr<ApplicationSensorConfig>> sensors() const {
222 std::vector<std::shared_ptr<ApplicationSensorConfig>> result;
223 ob_error *error = nullptr;
224 auto count = ob_application_config_get_count(handle_, OB_APP_CONFIG_SENSORS, &error);
225 Error::handle(&error);
226 for(uint32_t i = 0; i < count; ++i) {
227 auto sensorType = static_cast<OBSensorType>(ob_application_config_get_int(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_SENSOR_TYPE, i), &error));
228 Error::handle(&error);
229 auto sensorConfig = std::make_shared<ApplicationSensorConfig>(sensorType);
230 // enable
231 auto enable = ob_application_config_get_bool(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_STREAM_ENABLED, i), &error);
232 Error::handle(&error);
233 sensorConfig->enableStream(enable);
234 // profile
235 auto profile = ob_application_config_get_profile(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_STREAM_PROFILE, i), &error);
236 Error::handle(&error);
237 sensorConfig->setStreamProfile(StreamProfileFactory::create(profile));
238 result.emplace_back(sensorConfig);
239 // undistortion
240 enable = ob_application_config_get_bool(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_UNDISTORTION_ENABLED, i), &error);
241 Error::handle(&error, false); // optional
242 sensorConfig->enableUndistortion(enable);
243 }
244 return result;
245 }

253 void setSensor(const std::shared_ptr<ApplicationSensorConfig> &sensor) {
254 if(!sensor) {
255 return;
256 }
257 ob_error *error = nullptr;
258 auto index = sensorIndex(sensor->sensorType());
259 // enable
260 ob_application_config_set_bool(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_STREAM_ENABLED, index), sensor->isStreamEnabled(), &error);
261 Error::handle(&error);
262 // profile
263 auto profile = sensor->streamProfile();
264 if(!profile) {
265 throw std::runtime_error("ApplicationSensorConfig: stream profile must not be null");
266 }
267 ob_application_config_set_profile(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_STREAM_PROFILE, index), profile->getImpl(), &error);
268 Error::handle(&error);
269 // undistortion
270 ob_application_config_set_bool(handle_, OB_APP_CONFIG_ARRAY_ITEM(OB_APP_CONFIG_UNDISTORTION_ENABLED, index), sensor->isUndistortionEnabled(), &error);
271 Error::handle(&error);
272 }

385 uint32_t sensorIndex(OBSensorType sensorType) const {
386 ob_error *error = nullptr;
387 auto count = ob_application_config_get_count(handle_, OB_APP_CONFIG_SENSORS, &error);
388 Error::handle(&error);
389 for(uint32_t i = 0; i < count; ++i) {
