1312// Define the is() template function for the Frame class
1313template <typename T> bool Frame::is() const {
1314 switch(this->getType()) {
1315 case OB_FRAME_IR_LEFT: // Follow
1316 case OB_FRAME_IR_RIGHT: // Follow
1317 case OB_FRAME_IR:
1318 return (typeid(T) == typeid(IRFrame) || typeid(T) == typeid(VideoFrame));
1319 case OB_FRAME_DEPTH:
1320 return (typeid(T) == typeid(DepthFrame) || typeid(T) == typeid(VideoFrame));
1321 case OB_FRAME_COLOR:
1322 case OB_FRAME_COLOR_LEFT:
1323 case OB_FRAME_COLOR_RIGHT:
1324 return (typeid(T) == typeid(ColorFrame) || typeid(T) == typeid(VideoFrame));
1325 case OB_FRAME_CONFIDENCE:
1326 return (typeid(T) == typeid(ConfidenceFrame) || typeid(T) == typeid(VideoFrame));
1327 case OB_FRAME_GYRO:
1328 return (typeid(T) == typeid(GyroFrame));
1329 case OB_FRAME_ACCEL:
1330 return (typeid(T) == typeid(AccelFrame));
1331 case OB_FRAME_POINTS:
1332 return (typeid(T) == typeid(PointsFrame));
1333 case OB_FRAME_LIDAR_POINTS:
1334 return (typeid(T) == typeid(LiDARPointsFrame));
1335 case OB_FRAME_SET:
1336 return (typeid(T) == typeid(FrameSet));
1337 default:
1338 std::cout << "ob::Frame::is() did not catch frame type: " << (int)this->getType() << std::endl;
1339 break;
1340 }
1341 return false;
1342}
